import Item from '../../../components/mc/Item.astro';
import { meta, worldgen, seasons, drinks, nameOf, list } from '../../../lib/l10n/uk';
import compat from '../../../data/gen/compat.json';
import { optional, downloads, gaps, neoforgeVersions, forgeVersions } from '../../../data/code/downloads';

export const bopNames = { 'biomesoplenty:field': 'Поле', 'biomesoplenty:pasture': 'Пасовище', 'biomesoplenty:prairie': 'Прерія', 'biomesoplenty:mediterranean_forest': 'Середземноморський ліс', 'biomesoplenty:orchard': 'Сад', 'biomesoplenty:woodland': 'Лісисті землі', 'biomesoplenty:seasonal_forest': 'Сезонний ліс', 'biomesoplenty:lavender_field': 'Лавандове поле', 'biomesoplenty:lavender_forest': 'Лавандовий ліс' };
export const bop = worldgen.filter((w) => w.extra).map((w) => ({ crop: w.crop, biomes: w.optional.map((b) => bopNames[b.id] ?? b.name) }));
export const seasonNames = { spring: 'весна', summer: 'літо', autumn: 'осінь', winter: 'зима' };
export const seasonRows = Object.entries(seasons);
export const v = compat.versions;
// Wheat beer feeds through Saturation: (level + 1) food and twice that in saturation per tick it lasts
export const beer = drinks.filter((d) => d.effect.id === 'minecraft:saturation').map((d) => ({ ...d, food: (d.effect.amplifier + 1) * d.effect.duration }));
export const young = beer.find((d) => d.stage === 'young');
export const aged = beer.find((d) => d.stage === 'aged');
// A row of the thirst table is all young drinks, all aged drinks, or a few named items
export const stageOf = (id) => drinks.find((d) => d.id === id)?.stage;
export const rowLabel = (items) => {
	const all = (stage) => items.every((id) => stageOf(id) === stage) && items.length === drinks.filter((d) => d.stage === stage).length;
	return all('young') ? 'Кожен молодий напій' : all('aged') ? 'Кожен витриманий напій' : null;
};
export const groups = { fruits: 'Фрукти', carbs: 'Вуглеводи', proteins: 'Білки', vegetables: 'Овочі', sugars: 'Цукри' };

Vine & Barrel {meta.version} працює на **Minecraft {list(neoforgeVersions)}** із **NeoForge** або **Fabric**, а на
**Minecraft {list(forgeVersions)}** — із **Forge** або **Fabric**. У розділі [Встановлення](/getting-started/#встановлення) є файл для кожної версії та завантажувач, який він потребує. Інші моди йому не потрібні.
Усе нижче необов'язкове й вмикається саме, коли встановлено відповідний мод:

- **NeoForge:** {optional.neoforge.join(', ')}.
- **Forge:** {optional.forge.join(', ')}.
- **Fabric:** {optional.fabric.join(', ')}.

Деяких із них немає для частини версій Minecraft. Де вони працюють:

Serene Seasons і Biomes O' Plenty працюють через теги.

## Farmer's Delight

Тепло плити, лози на мотузках (як на шпалері), родючий ґрунт, напої з пляшок FD, кілька страв із каструлі, бенкет і хліб. Дивись [Farmer's Delight](/farmers-delight/).

## Kaleidoscope Tavern

З **Kaleidoscope Tavern** (на Fabric — **Kaleidoscope Tavern Refabricated**) виноград працює в обидва боки. Його
виноград вважається нашим <Item id="red_grapes" />, а зелений виноград — нашим <Item id="white_grapes" />, тож варильний казан варить із них <Item id="grape_juice" />
і <Item id="white_grape_juice" /> так само, як із нашого. В інший бік його давильний чан вичавлює наш червоний виноград у свій виноградний
сік, а білий — у свій сік із зеленого винограду. Його крижаний і золотий виноград лишається тільки його.

Кожен мод має свої напої та своє сп'яніння: від напоїв Kaleidoscope ти не будеш **Напідпитку**, а наші не додаються до
його сп'яніння.

## Towns & Towers

Towns & Towers лишає звичайні села разом із корчмами на рівнинах, у пустелях, саванах, сніжних рівнинах і тайзі, а всюди
інде будує свої. У них теж буває [корчма](/villages/), приблизно в кожному третьому селі, як у звичайних, у стилі найближчого
до них звичайного села: швейцарським селам на луках і селам у березових, квіткових та інших лісах дістається корчма рівнин,
польським селам старої тайги тайгова, селам вікінгів у сніжній тайзі сніжна, селам безплідних земель пустельна, а селам
джунглів корчма савани. Так само і його селам для біомів Terralith та інших модів, збудованим так само. У його селах на пляжі,
у гаю, на сніжних схилах, на плато савани, на соняшниковій рівнині, на болоті й на грибних полях немає ділянки, куди вмістилася
б корчма, тож її там немає. Усе працює однаково, чи Towns & Towers стоїть модом, чи датапаком.

## JEI

JEI показує категорії рецептів **Варильний казан** і **Діжка для витримки**, з часом варіння та днями, які займає кожен етап,
а також сторінки з описом діжки й диких культур. Зібрано під JEI {meta.jei}.

## Jade

Подивись на казан чи діжку, щоб побачити, що всередині: варево чи напій, скільки порцій лишилося, наскільки просунулося варіння, чи
нагрітий казан і на якому етапі діжка. Зібрано під Jade {meta.jade.split('+')[0]}.

## AppleSkin

Підказка про їжу та попередній перегляд шкали голоду з AppleSkin працюють для всього, що можна з'їсти чи випити. Соки й страви показують
власну поживність. <Item id={young.id} /> годує через свій бонус насичення, тож AppleSkin показує, що цей бонус заповнює: **{young.food}**
голоду і **{young.food * 2}** насичення, або **{aged.food}** і **{aged.food * 2}** для <Item id={aged.id} />. Решта
вин, пив, сидрів і медовух не годує, тож їжі вони не показують. Зібрано під AppleSkin {v.appleskin}.

## Спрага

З **Tough As Nails**, **Homeostatic** чи **Thirst Was Taken 2** напої втамовують спрагу: соки багато, вина, пиво,
сидри й медовуха трохи, витримані трохи більше, ніж молоді. У модах, що враховують і їжу, тушковані страви, юшка й півень у вині рахуються
як рагу, а виноград, червоний чи білий, як солодкі ягоди.

Taken 2 {v.thirstwastaken2}.

## Nutritional Balance

Лише NeoForge. Кожна їжа й кожен напій зараховується до харчових груп Nutritional Balance:

<ul>
	{Object.entries(compat.nutrients).map(([group, ids]) => (
		<li><strong>{groups[group] ?? group}:</strong> {ids.map((id, i) => <>{i > 0 && ', '}<Item id={id} /></>)}</li>
	))}
</ul>

Вина, пиво, сидри й медовуха самі по собі не мають харчової цінності (пшеничне пиво насичує через свій бонус насичення, який ці моди не враховують), тож твої групи наповнюють лише виноград, соки й страви. Сирий ячмінь рахується
як вуглеводи, тож усе, що інший мод робить із нього, теж. Перевірено з Nutritional Balance {v.nutritionalbalance}.

Кожен напій і кожна страва також мають спільні теги обох завантажувачів (напої, соки, фрукти, супи), тож інші моди, що їх читають, бачать
їх такими, якими вони є.

## Spice of Life Onion

Працює як є. Страви зараховуються до різноманітності твоєї їжі, як і будь-яка їжа. Напої самі по собі не мають харчової цінності (бонус насичення пшеничного пива
не рахується), тож, як і напої Farmer's Delight, різноманітності не додають. Перевірено з Spice of Life Onion {v.solonion}.

## Biomes O' Plenty

Дикі культури ростуть у кожному біомі з тегом лісу, рівнини, савани, річки чи гірського схилу (де температура їм підходить), тож модові біоми з такими тегами отримують їх автоматично.
З Biomes O' Plenty вони також ростуть у кількох біомах, які їм пасують:

<ul>
	{bop.map((b) => <li><Item id={b.crop} />: {list(b.biomes)}</li>)}
</ul>

## Serene Seasons

З Serene Seasons кожна культура росте у свої сезони:

<ul>
	{['vineandbarrel:red_grapes', 'vineandbarrel:white_grapes', 'vineandbarrel:hops', 'vineandbarrel:barley_seeds'].map((id) => (
		<li><Item id={id} />: {list(seasonRows.filter(([, ids]) => ids.includes(id)).map(([s]) => seasonNames[s] ?? s))}</li>
	))}
</ul>
