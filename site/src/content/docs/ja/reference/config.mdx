import { Aside } from '@astrojs/starlight/components';
import { rules, LOCALE } from '../../../../lib/l10n/ja';
import { configFiles } from '../../../../data/code/config';

export const exported = rules.tipsy.config;
export const value = (o) => String(exported[o.key] ?? o.fallback);
export const rows = (f) => f.options.map((o) => ({ ...o, def: value(o), what: typeof o.what === 'function' ? o.what(rules.tipsy, LOCALE) : o.what }));
export const wild = configFiles.worldgen.options.map((o) => ({ ...o, def: String(rules.crops.config[o.key] ?? o.fallback), what: o.what(rules.tipsy, LOCALE) }));

Vine & Barrelの設定ファイルは2つあります。ひとつはサーバー用で、[ほろ酔い](/tipsy/)のルールと野生の作物ごとのスイッチが入っています。
もうひとつはプレイヤーごとのもので、ほろ酔いが自分の画面と耳にどう効くかを決めます。それ以外、レシピから樽の熟成にかかる時間まで、
すべてMODの出荷時のまま動き、ほかのレシピと同じようにデータパックで変更できます（Minecraft 1.21.1 と 1.20.1 では、村人の取引だけは変更できません）。

ゲーム内では設定画面で変更します。NeoForgeでは**Mod**、Vine & Barrel、**コンフィグ**の順に開きます。Fabricでは
[Mod Menu](https://modrinth.com/mod/modmenu)を入れると同じボタンが出ます。ここでの変更はすぐに反映されます。サーバーのオプションは、
シングルプレイでは(ワールドをLANに公開していなければ)ここに表示されます。サーバーでは、サーバー側の値が使われます。Forge(Minecraft 1.20.1)にはこのボタンがないので、
ファイルを直接編集してください。

ファイルそのものについて: NeoForgeとForgeではTOMLファイルで、オプションは<code>[{configFiles.server.section}]</code>の下にあります。
Fabricでは<code>config</code>フォルダにある普通の<code>.properties</code>ファイルで、同じオプションを
<code>{configFiles.server.section}.option=value</code>の形で書きます。Fabricのクライアントファイルはゲームの起動時に読み込まれるので、
編集したら再起動してください。サーバーファイルは<code>/reload</code>で読み込み直されます。野生の作物のスイッチは
<code>[{configFiles.worldgen.section}]</code>の下にあります（Fabricでは<code>{configFiles.worldgen.section}.option=value</code>）。

## サーバーオプション

NeoForgeとForgeでは<code>{configFiles.server.file}</code>、Fabricでは<code>{configFiles.server.fabricFile}</code>にあります。サーバーの
値は参加するすべてのプレイヤーに送られるので、全員が同じルールで遊びます。ワールドごとに別の値にもできます。そのワールドの
<code>serverconfig</code>フォルダにファイルを入れると、そのワールドが動いているあいだは<code>config</code>にあるものより優先されます。

<table>
	<thead><tr><th>オプション</th><th>初期値</th><th>効果</th></tr></thead>
	<tbody>
		{rows(configFiles.server).map((o) => (
			<tr id={`opt-${o.key}`} class="ref-target">
				<td><code>{o.key}</code></td>
				<td><code>{o.def}</code>{o.range && <><br /><span class="soft">{o.range[0]}から{o.range[1]}</span></>}</td>
				<td>{o.what.split('`').map((part, i) => (i % 2 ? <code>{part}</code> : part))}</td>
			</tr>
		))}
	</tbody>
</table>

<Aside title="ノンアルコールモード">
`enabled = false`(Fabricでは`tipsy.enabled=false`)にすると、飲み物はただの強化効果つきの飲み物になります。ほろ酔いも、しゃっくりも、記憶飛びもありませんが、名前と強化効果は同じです。
家族向けのサーバーや、ふらふらが好きではない人に向いています。
</Aside>

### 野生の作物

同じファイルの<code>[{configFiles.worldgen.section}]</code>の下に、[野生の作物](/crops/#野生の作物)ごとのスイッチがあります。変わるのはこれから生成される
土地だけです。オフにすると新しいチャンクにはその作物が生えず、すでに外に生えている株はそのまま残ります。

<table>
	<thead><tr><th>オプション</th><th>初期値</th><th>効果</th></tr></thead>
	<tbody>
		{wild.map((o) => (
			<tr id={`opt-${o.key}`} class="ref-target">
				<td><code>{o.key}</code></td>
				<td><code>{o.def}</code></td>
				<td>{o.what.split('`').map((part, i) => (i % 2 ? <code>{part}</code> : part))}</td>
			</tr>
		))}
	</tbody>
</table>

<Aside title="ほかのMODにもうブドウがある？">
パックのMODが自前の野生のブドウ、ホップ、大麦を生やすなら、こちらのものをオフにしましょう（`wildWhiteGrapes = false`、
Fabricでは`worldgen.wildWhiteGrapes=false`）。同じ植物が2種類見つかって迷うことがなくなります。村の畑では
これまでどおり育ち、チェストや商人も種を持っているので、作物がゲームから消えることはありません。
</Aside>

## クライアントオプション

NeoForgeとForgeでは<code>{configFiles.client.file}</code>、Fabricでは<code>{configFiles.client.fabricFile}</code>にあります。プレイヤーが
各自で設定するもので、見え方だけが変わります。

<table>
	<thead><tr><th>オプション</th><th>初期値</th><th>効果</th></tr></thead>
	<tbody>
		{rows(configFiles.client).map((o) => (
			<tr id={`opt-${o.key}`} class="ref-target">
				<td><code>{o.key}</code></td>
				<td><code>{o.def}</code>{o.range && <><br /><span class="soft">{o.range[0]}から{o.range[1]}</span></>}</td>
				<td>{o.what.split('`').map((part, i) => (i % 2 ? <code>{part}</code> : part))}</td>
			</tr>
		))}
	</tbody>
</table>

<Aside type="tip" title="画面酔いが心配?">
「酔いの画面」を0%にしましょう(`drunkScreen = 0`、Fabricでは`tipsy.drunkScreen=0`)。ほろ酔いが勝手に画面を動かすことはなく、0にすれば、サーバーの設定にかかわらず、しゃっくりやよろめき、つまずきでも画面は動きません。バニラの「画面の歪み」スライダーでも弱められ、「暗闇の脈動」スライダーでは居眠りのときに下がってくるまぶたを弱められます。つまずいて顔から倒れる姿は画面効果ではないので、そのまま表示されます。物が二重に見えるのがつらければ、「二重に見える」をオフにしましょう(`doubleVision = false`)。ほろ酔いのそれ以外はそのままです。
</Aside>
