import Item from '../../../components/mc/Item.astro';
import { meta, worldgen, seasons, drinks, nameOf, list } from '../../../lib/l10n/zh-cn';
import compat from '../../../data/gen/compat.json';
import { optional, downloads, gaps } from '../../../data/code/downloads';

export const bop = worldgen.filter((w) => w.extra).map((w) => ({ crop: w.crop, biomes: w.optional.map((b) => b.name) }));
export const seasonRows = Object.entries(seasons);
export const seasonName = { spring: '春季', summer: '夏季', autumn: '秋季', winter: '冬季' };
export const v = compat.versions;
// Wheat beer feeds through Saturation: (level + 1) food and twice that in saturation per tick it lasts
export const beer = drinks.filter((d) => d.effect.id === 'minecraft:saturation').map((d) => ({ ...d, food: (d.effect.amplifier + 1) * d.effect.duration }));
export const young = beer.find((d) => d.stage === 'young');
export const aged = beer.find((d) => d.stage === 'aged');
// A row of the thirst table is all young drinks, all aged drinks, or a few named items
export const stageOf = (id) => drinks.find((d) => d.id === id)?.stage;
export const rowLabel = (items) => {
	const all = (stage) => items.every((id) => stageOf(id) === stage) && items.length === drinks.filter((d) => d.stage === stage).length;
	return all('young') ? '所有新酿饮品' : all('aged') ? '所有陈酿饮品' : null;
};
export const groups = { fruits: '水果', carbs: '碳水化合物', proteins: '蛋白质', vegetables: '蔬菜', sugars: '糖类' };

Vine & Barrel {meta.version} 运行在 **Minecraft {list(downloads.map((r) => r.minecraft))}** 上，搭配 **NeoForge** 或 **Fabric**。每个版本的文件和所需的加载器见[安装](/getting-started/#安装)。它不需要其他模组。下面的一切都是可选的，装上对应的模组就会自动启用：

- **NeoForge**：{optional.neoforge.join('、')}。
- **Fabric**：{optional.fabric.join('、')}。

其中几个模组并没有为每个 Minecraft 版本发布。可用的版本和加载器如下：

Serene Seasons 和 Biomes O' Plenty 通过标签起作用。

## Farmer's Delight

炉灶热源、绳子上的藤蔓（和棚架上一样）、沃土、用 FD 的瓶子酿的饮品，以及好几道厨锅菜肴、一道盛宴和一种面包。见 [Farmer's Delight](/farmers-delight/)。

## JEI

JEI 会显示**酿造锅**和**陈酿桶**两个配方分类，包括酿造时间和每个阶段需要的天数，还有陈酿桶和野生作物的信息页。基于 JEI {meta.jei} 开发。

## Jade

看向酿造锅或陈酿桶，就能看到里面有什么：酒或饮品、剩余的份数、酿造进度、酿造锅是否受热，以及陈酿桶处于哪个阶段。基于 Jade {meta.jade.split('+')[0]} 开发。

## AppleSkin

AppleSkin 的食物提示和饥饿条预览，对所有能吃能喝的东西都有效。果汁和菜肴显示它们自己的食物值。<Item id={young.id} />是通过它的饱和增益来填饱肚子的，所以 AppleSkin 显示的是这个增益填充的量：**{young.food}** 点饥饿值和 **{young.food * 2}** 点饱和度，<Item id={aged.id} />则是 **{aged.food}** 和 **{aged.food * 2}**。其他葡萄酒、啤酒、苹果西打和蜂蜜酒不提供食物，所以不显示食物值。基于 AppleSkin {v.appleskin} 开发。

## Thirst

装有 **Tough As Nails**、**Homeostatic** 或 **Thirst Was Taken 2** 时，这些饮品可以解渴：果汁解渴很多，葡萄酒、啤酒、苹果西打和蜂蜜酒解渴一点，陈酿的比新酿的稍多一些。在同时统计食物的模组里，炖菜、肉汤和红酒炖鸡算作炖菜，红葡萄和白葡萄算作甜浆果。

	<thead>

Tough As Nails 只统计饮品。已用 Tough As Nails {v.toughasnails}、Homeostatic {v.homeostatic} 和 Thirst Was Taken 2 {v.thirstwastaken2} 测试。

## Nutritional Balance

仅限 NeoForge。每种食物和饮品都计入 Nutritional Balance 的食物类别：

<ul>
	{Object.entries(compat.nutrients).map(([group, ids]) => (
		<li><strong>{groups[group] ?? group}</strong>：{ids.map((id, i) => <>{i > 0 && '、'}<Item id={id} /></>)}</li>
	))}
</ul>

葡萄酒、啤酒、苹果西打和蜂蜜酒本身没有食物值（小麦啤酒通过它的饱和增益填饱肚子，而这些模组不统计这个），所以只有葡萄、果汁和菜肴会填充你的各个类别。生大麦算作碳水化合物，因此其他模组用它做出的东西也一样。已用 Nutritional Balance {v.nutritionalbalance} 测试。

每种饮品和食物还带有两个加载器共用的常见标签（饮品、果汁、水果、汤），所以读取这些标签的其他模组能正确认出它们。

## Spice of Life Onion

直接可用。菜肴和任何食物一样，计入你的食物多样性。饮品本身没有食物值（小麦啤酒的饱和增益不计入），所以和 Farmer's Delight 的饮品一样，不增加多样性。已用 Spice of Life Onion {v.solonion} 测试。

## Biomes O' Plenty

野生作物生长在所有带有森林、平原、热带草原、河流或山坡标签的生物群系里（只要温度合适），所以带有这些标签的模组生物群系会自动长出它们。装有 Biomes O' Plenty 时，它们还会长在几个适合它们的生物群系里：

<ul>
	{bop.map((b) => <li><Item id={b.crop} />：{list(b.biomes)}</li>)}
</ul>

## Serene Seasons

装有 Serene Seasons 时，每种作物在各自的季节里生长：

<ul>
	{['vineandbarrel:red_grapes', 'vineandbarrel:white_grapes', 'vineandbarrel:hops', 'vineandbarrel:barley_seeds'].map((id) => (
		<li><Item id={id} />：{list(seasonRows.filter(([, ids]) => ids.includes(id)).map(([s]) => seasonName[s] ?? s))}</li>
	))}
</ul>
