import Item from '../../../components/mc/Item.astro';
import { meta, worldgen, seasons, drinks, nameOf, list } from '../../../lib/l10n/ko';
import compat from '../../../data/gen/compat.json';
import { optional, downloads, gaps, neoforgeVersions, forgeVersions } from '../../../data/code/downloads';

export const bop = worldgen.filter((w) => w.extra).map((w) => ({ crop: w.crop, biomes: w.optional.map((b) => b.name) }));
export const seasonRows = Object.entries(seasons);
export const v = compat.versions;
// Wheat beer feeds through Saturation: (level + 1) food and twice that in saturation per tick it lasts
export const beer = drinks.filter((d) => d.effect.id === 'minecraft:saturation').map((d) => ({ ...d, food: (d.effect.amplifier + 1) * d.effect.duration }));
export const young = beer.find((d) => d.stage === 'young');
export const aged = beer.find((d) => d.stage === 'aged');
// A row of the thirst table is all young drinks, all aged drinks, or a few named items
export const stageOf = (id) => drinks.find((d) => d.id === id)?.stage;
export const rowLabel = (items) => {
	const all = (stage) => items.every((id) => stageOf(id) === stage) && items.length === drinks.filter((d) => d.stage === stage).length;
	return all('young') ? '모든 어린 술' : all('aged') ? '모든 숙성된 술' : null;
};
export const groups = { fruits: '과일', carbs: '탄수화물', proteins: '단백질', vegetables: '채소', sugars: '당류' };

Vine & Barrel {meta.version}은 **Minecraft {list(neoforgeVersions)}**에서 **NeoForge** 또는 **Fabric**으로,
**Minecraft {list(forgeVersions)}**에서 **Forge** 또는 **Fabric**으로 실행돼요. 버전별 파일과 필요한 로더는 [설치](/getting-started/#설치)에 있어요. 다른 모드는 필요 없어요.
아래는 모두 선택 사항이고, 해당 모드가 설치되어 있으면 저절로 켜져요.

- **NeoForge:** {optional.neoforge.join(', ')}.
- **Forge:** {optional.forge.join(', ')}.
- **Fabric:** {optional.fabric.join(', ')}.

몇몇 모드는 일부 Minecraft 버전용 빌드가 없어요. 작동하는 곳은 다음과 같아요.

<table>
	<thead><tr><th></th>{downloads.map((r) => <th>{r.minecraft}</th>)}</tr></thead>
	<tbody>
		{gaps.map((g) => (
			<tr>
				<td>{g.mod}</td>
				{g.cells.map((c) => <td>{c}</td>)}
			</tr>
		))}
	</tbody>
</table>

Serene Seasons와 Biomes O' Plenty는 태그로 작동해요.

## Farmer's Delight

화덕 열, 밧줄 위의 덩굴(격자 지지대처럼), 비옥한 흙, FD 병으로 만드는 술, 조리솥 요리 몇 가지와 잔치 요리, 빵. [Farmer's Delight](/farmers-delight/)를 참고하세요.

## Kaleidoscope Tavern

**Kaleidoscope Tavern**(Fabric에서는 **Kaleidoscope Tavern Refabricated**)이 있으면 포도를 서로 쓸 수 있습니다. 그쪽 포도는
우리 <Item id="red_grapes" />, 그쪽 청포도는 우리 <Item id="white_grapes" />로 쳐서, 양조 솥이 우리 포도와 똑같이 <Item id="grape_juice" />와 <Item id="white_grape_juice" />를 만듭니다.
반대로 그쪽 압착 통은 우리 적포도를 그쪽 포도 주스로, 우리 청포도를 그쪽 청포도 주스로 짜냅니다. 얼음 포도와
황금 포도는 그쪽 것으로 남습니다.

음료와 취기는 모드마다 따로입니다. Kaleidoscope의 음료로는 **알딸딸함**이 되지 않고, 우리 음료도 그쪽 취기에 더해지지 않습니다.

## JEI

JEI에는 **양조 솥**과 **숙성 통** 레시피 분류가 있고, 양조 시간과 단계별 일수, 그리고 통과 야생 작물의 정보 페이지가 나와요.
JEI {meta.jei} 기준으로 만들었어요.

## Jade

솥이나 통을 보면 안에 뭐가 있는지 보여 줘요. 양조물이나 술, 남은 인분, 양조가 얼마나 진행됐는지, 솥이 데워졌는지, 통이 어느 단계인지요.
Jade {meta.jade.split('+')[0]} 기준으로 만들었어요.

## AppleSkin

AppleSkin의 음식 툴팁과 허기 게이지 미리보기는 먹거나 마실 수 있는 모든 것에 작동해요. 주스와 요리는 각자의 음식 값을 보여 줘요.
<Item id={young.id} />는 포화 버프로 배를 채우니까, AppleSkin은 그 버프가 채우는 만큼을 보여 줘요. 허기 **{young.food}**,
포화도 **{young.food * 2}**이고, <Item id={aged.id} />는 **{aged.food}**와 **{aged.food * 2}**예요. 다른 와인, 맥주, 사이더, 벌꿀술은
배를 채우지 않아서 음식 값이 나오지 않아요. AppleSkin {v.appleskin} 기준으로 만들었어요.

## 갈증

**Tough As Nails**, **Homeostatic**, **Thirst Was Taken 2**가 있으면 술이 갈증을 풀어 줘요. 주스는 많이, 와인, 맥주, 사이더, 벌꿀술은 조금,
숙성된 술은 어린 술보다 조금 더요. 음식도 세는 모드에서는 조림 요리와 수프, 코코뱅이 스튜처럼, 적포도와 청포도가 달콤한 열매처럼 쳐져요.

<table>
	<thead>
		<tr><th></th><th>Tough As Nails<br />갈증, 수분</th><th>Homeostatic<br />물, 포화도</th><th>Thirst Was Taken 2<br />갈증, 해소량</th></tr>
	</thead>
	<tbody>
		{compat.thirst.map((row) => (
			<tr>
				<td>{rowLabel(row.items) ?? row.items.map((id, i) => <>{i > 0 && ', '}<Item id={id} /></>)}</td>
				<td>{row.tan ? `${row.tan.thirst}, ${row.tan.hydration}%` : '–'}</td>
				<td>{row.homeostatic.amount}, {row.homeostatic.saturation.toFixed(1)}</td>
				<td>{row.twt ? `${row.twt.thirst}, ${row.twt.quenched}` : '–'}</td>
			</tr>
		))}
	</tbody>
</table>

Tough As Nails는 음료만 세요. Tough As Nails {v.toughasnails}, Homeostatic {v.homeostatic}, Thirst Was
Taken 2 {v.thirstwastaken2}로 테스트했어요.

## Nutritional Balance

NeoForge 전용이에요. 모든 음식과 음료가 Nutritional Balance의 식품군에 들어가요.

<ul>
	{Object.entries(compat.nutrients).map(([group, ids]) => (
		<li><strong>{groups[group] ?? group}:</strong> {ids.map((id, i) => <>{i > 0 && ', '}<Item id={id} /></>)}</li>
	))}
</ul>

와인, 맥주, 사이더, 벌꿀술은 자체 음식 값이 없어서(밀맥주는 포화 버프로 배를 채우는데, 이 모드들은 그걸 세지 않아요) 포도, 주스, 요리만 식품군을 채워요. 생 보리는
탄수화물로 치니까 다른 모드가 보리로 만든 것도 마찬가지예요. Nutritional Balance {v.nutritionalbalance}로 테스트했어요.

모든 술과 음식에는 두 로더가 공통으로 쓰는 태그(음료, 주스, 과일, 수프)도 붙어 있어서, 그 태그를 읽는 다른 모드가 제대로 알아봐요.

## Spice of Life Onion

그대로 작동해요. 요리는 다른 음식처럼 음식 다양성에 들어가요. 술은 자체 음식 값이 없어서(밀맥주의
포화 버프는 세지 않아요) Farmer's Delight의 음료처럼 다양성에 더해지지 않아요. Spice of Life Onion {v.solonion}으로 테스트했어요.

## Biomes O' Plenty

야생 작물은 숲, 평원, 사바나, 강, 산비탈 태그가 붙은 모든 바이옴에서 자라요(기온이 맞을 때). 그래서 그 태그가 있는 모드 바이옴에는 자동으로 생겨요.
Biomes O' Plenty가 있으면 어울리는 바이옴 몇 곳에서도 자라요.

<ul>
	{bop.map((b) => <li><Item id={b.crop} />: {list(b.biomes)}</li>)}
</ul>

## Serene Seasons

Serene Seasons가 있으면 작물마다 자기 계절에 자라요.

<ul>
	{['vineandbarrel:red_grapes', 'vineandbarrel:white_grapes', 'vineandbarrel:hops', 'vineandbarrel:barley_seeds'].map((id) => (
		<li><Item id={id} />: {list(seasonRows.filter(([, ids]) => ids.includes(id)).map(([s]) => s))}</li>
	))}
</ul>
